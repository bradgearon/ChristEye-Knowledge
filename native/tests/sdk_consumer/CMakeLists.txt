cmake_minimum_required(VERSION 3.20)
project(AnaieroIndependentSdkConsumer LANGUAGES C CXX)
find_package(Anaiero 1 REQUIRED CONFIG)
enable_testing()
add_library(independent_native MODULE provider.cpp)
target_compile_definitions(independent_native PRIVATE ANAIERO_MODULE_BUILD)
target_link_libraries(independent_native PRIVATE anaiero::anaiero_module_api)
add_library(independent_invalid MODULE invalid_module.cpp)
target_compile_definitions(independent_invalid PRIVATE ANAIERO_MODULE_BUILD)
target_link_libraries(independent_invalid PRIVATE anaiero::anaiero_contracts)
add_executable(independent_process provider.cpp)
target_link_libraries(independent_process PRIVATE anaiero::anaiero_module_api)
add_executable(independent_consumer consumer.cpp)
target_link_libraries(independent_consumer PRIVATE anaiero::anaiero_runtime anaiero::anaiero_experiments
  anaiero::anaiero_module_host anaiero::anaiero_observers anaiero::anaiero_storage_native Threads::Threads)
add_library(independent_c_header OBJECT c_header.c)
target_include_directories(independent_c_header PRIVATE ${ANAIERO_SDK_INCLUDE_DIR})
target_link_libraries(independent_c_header PRIVATE anaiero::anaiero_contracts)
foreach(target independent_native independent_invalid independent_process independent_consumer)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wconversion -Werror)
  endif()
endforeach()
add_test(NAME installed_sdk_conformance COMMAND independent_consumer
  $<TARGET_FILE:independent_native> $<TARGET_FILE:independent_process> $<TARGET_FILE:independent_invalid>
  ${CMAKE_CURRENT_BINARY_DIR}/independent-store)
